import streamlit as st
import json
from groq import Groq

st.set_page_config(page_title="Recall AI", page_icon="⏰", layout="centered")

st.title("⏰ Recall AI")
st.markdown("### Turn messy WhatsApp chats into instant reminders using Meta Llama")
st.write("---")

# Temporary free engine key for testing while your Meta credits process
GROQ_API_KEY = "PASTE_YOUR_GROQ_API_KEY_HERE" 
client = Groq(api_key=GROQ_API_KEY)

user_input = st.text_area(
    "💬 Paste your raw WhatsApp text message here:",
    placeholder="e.g., Hey bro, let's lock in the project review sync this coming Sunday at 4 PM.",
    height=120
)

if st.button("🚀 Analyze & Extract Details", type="primary"):
    if not user_input.strip():
        st.warning("⚠️ Please paste a message first!")
    else:
        with st.spinner("🧠 Meta Llama is parsing the conversation..."):
            prompt = f"""
            You are Recall AI, a precise scheduling assistant.
            Analyze this message: "{user_input}"
            
            Extract and return ONLY a valid, raw JSON object with these exact keys. Do not include markdown code block syntax:
            - task_name: (string summary of the event)
            - date: (YYYY-MM-DD format or null)
            - time: (HH:MM format or null)
            - ambiguity_flag: (true if the text is vague, false if clear)
            - response_to_user: (a direct, short note confirming or asking to clarify details)
            """
            
            try:
                chat_completion = client.chat.completions.create(
                    messages=[{"role": "user", "content": prompt}],
                    model="llama-3.1-8b-instant",
                )
                
                raw_content = chat_completion.choices.message.content.strip()
                if raw_content.startswith("json"):
                    raw_content = raw_content[7:-3].strip()
                elif raw_content.startswith(""):
                    raw_content = raw_content[3:-3].strip()
                
                ai_data = json.loads(raw_content)
                
                st.success("✨ Extraction Successful!")
                st.write("### 📋 Edit & Confirm Event Details")
                
                edited_title = st.text_input("🔹 Event Title:", value=ai_data.get("task_name", "New Reminder"))
                edited_date = st.text_input("📅 Extracted Date:", value=ai_data.get("date", ""))
                edited_time = st.text_input("🕒 Extracted Time:", value=ai_data.get("time", ""))
                
                if ai_data.get("ambiguity_flag") is True:
                    st.info(f"🤔 *Clarification Needed:* {ai_data.get('response_to_user')}")
                else:
                    st.success(f"✅ *Llama Confidence High:* {ai_data.get('response_to_user')}")
                
                st.write("---")
                
                # Fast Google Calendar URL Generator Link
                import urllib.parse
                cal_title = urllib.parse.quote(edited_title)
                cal_date = edited_date.replace("-", "") if edited_date else "20261101"
                cal_time = edited_time.replace(":", "") if edited_time else "1200"
                
                gcal_url = f"https://google.com{cal_title}&dates={cal_date}T{cal_time}00Z/{cal_date}T{cal_time}00Z"
                
                st.markdown(f'<a href="{gcal_url}" target="_blank"><button style="background-color:#FF4B4B; color:white; border:none; padding:10px 20px; border-radius:5px; cursor:pointer;">📅 Add to Google Calendar</button></a>', unsafe_allow_html=True)
                    
            except Exception as e:
                st.error(f"Execution Error: {e}")
